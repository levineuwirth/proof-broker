\noindent
This card summarizes schemas, field names, tier semantics, and the
capability vocabulary defined in the v1.0 specification. Read alongside
the spec; this document does not motivate, only enumerates.

\section{Pipeline at a glance}

\begin{center}
\small
\fbox{Source} $\to$ \fbox{IR} $\to$ \fbox{Rewriter} $\to$ \fbox{Dispatcher} $\to$ \fbox{Adapter} $\to$ \fbox{Backend}
\end{center}

\noindent
Backend response flows back: \fbox{Certificate} $\to$ \fbox{Verifier} $\to$ \fbox{Lifting} $\to$ \fbox{Home-system proof term}

\small
\begin{tabular}{@{}lll@{}}

\section{Tier payloads}

\textbf{Tier 0:} \texttt{\{claim, backend\_attestation\}}.

\textbf{Tier 1:} \texttt{\{witness\_kind, witness\_data, checking\_recipe\}}.
Kinds: \texttt{farkas}, \texttt{sat\_assignment},
\texttt{sat\_unsat\_core}, \texttt{polynomial\_positivstellensatz}.

\textbf{Tier 2:} \texttt{\{lemmas\_used, strategy\_hint, structural\_hint\}}.
Fallback tier; probabilistic completeness.

\textbf{Tier 3:} \texttt{\{trace\_format, trace\_data,
trace\_dialect\_features, trace\_annotations\}}. Strategy stack:
native checker $\to$ cross-system replay $\to$ LLM-assisted $\to$ lemma extraction.

\section{Rewriting passes (v1)}

\noindent\texttt{propositional\_simplification} (default on),
\texttt{definition\_unfolding} (selective by concept tag),
\texttt{quotient\_elimination},
\texttt{equality\_saturation},
\texttt{quantifier\_prenexing},
\texttt{skolemization}.

\section{Verifier pre-tier checks}

\begin{enumerate}
  \item Envelope schema and version valid.
  \item \texttt{dispatch\_context\_hash} matches IR.
  \item \texttt{rewrite\_trace\_hash} matches trace.
  \item Backend recognized, not blocklisted.
  \item Library provenance hashes match home system.
\end{enumerate}

Any failure $\Rightarrow$ \texttt{Rejected(envelope\_invalid)}.

\section{Lifting phase}

Inverts in this order:
\begin{enumerate}
  \item Refinement record specializations (in reverse).
  \item Rewrite trace passes (in reverse).
\end{enumerate}

Output: home-system proof term of original goal.
Failure $\Rightarrow$ \texttt{Rejected(lifting\_failed)}.

\section{Cache key}

\begin{lstlisting}[style=plain]
(goal_hash,
 dispatch_context_hash,
 rewrite_trace_hash,
 backend_id,
 backend_version)
\end{lstlisting}

\textbf{Invalidate on:} backend version change, lemma content hash
change, home-system version change, manual.

\section{Dispatch failure modes}

\begin{description}
  \item[backend timeout] \texttt{unknown}, reason \texttt{timeout}
  \item[backend error] \texttt{error}, with diagnostic
  \item[verifier reject] logged with reason; speculative dispatches continue
  \item[reconstruction fail] per-strategy report
  \item[lifting fail] dispatch failed; no retry on same backend
\end{description}

\section{Tier-1 witness kinds (v1)}

\begin{description}
  \item[\texttt{farkas}] nonneg coefficients to LIA contradiction
  \item[\texttt{sat\_assignment}] variable-to-Boolean map
  \item[\texttt{sat\_unsat\_core}] minimal unsat clause subset
\end{description}

\section{Tier-3 trace formats (v1)}

\begin{description}
  \item[\texttt{alethe-2024}] CVC5 SMT proof (Lean replayer)
  \item[\texttt{tstp-fof}] Vampire FOL proof
  \item[\texttt{tstp-thf}] Vampire higher-order proof
  \item[\texttt{lean-tactic-script}] LLM-prover output
\end{description}

\section{Soundness invariants}

\begin{enumerate}
  \item Tier 1, 2, 3, 4 never expand trusted base; kernel re-checks every produced term.
  \item Tier 0 expands trust to backend; explicit per-project opt-in required.
  \item LLMs only propose; kernel disposes.
  \item Theory tags require verified embedding lemmas.
  \item Refinement records must be invertible.
  \item Rewrite passes preserve provability.
\end{enumerate}

\section{V1 scope checklist}

\noindent
\textbf{Home systems:} Lean 4 (full), Rocq (Tier 0+1 probe).
\quad
\textbf{Backends:} CVC5 (SMT), Vampire (ATP), generic LLM, ITP-via-Tier-0/2.
\quad
\textbf{Tiers:} 0 (gated), 1 (Farkas + SAT), 2 (fallback, demotable), 3 (Alethe + TPTP).
\quad
\textbf{Rewriting:} configurable pipeline, default = propositional simplification + selective unfolding.
\quad
\textbf{Excluded from v1:} Tier 4, ML premise selection, conversion metadata, full Rocq, HITs, mutual recursion in defn metadata, cross-ITP Tier 3.
